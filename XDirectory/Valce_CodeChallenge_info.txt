import org.junit.Test;

        List<Integer> expected = Arrays.asList(2, 5, 55, 69, 666, 2019, 42, 21, 9);
        assertEquals(expected, CustomLogic.removeDuplicates(input));
    }

    @Test
    public void testRemoveDuplicatesStrings() {
        List<String> input = Arrays.asList("John", "Tina", "John", "Bob");
        List<String> expected = Arrays.asList("John", "Tina", "Bob");
        assertEquals(expected, CustomLogic.removeDuplicates(input));
    }

    @Test
    public void testSortWithoutDuplicatesIntegers() {
        List<Integer> input = Arrays.asList(5, 2, 2, 9, 5);
        List<Integer> expected = Arrays.asList(2, 5, 9);
        assertEquals(expected, CustomLogic.sortWithoutDuplicates(input));
    }

    @Test
    public void testSortWithoutDuplicatesStrings() {
        List<String> input = Arrays.asList("Tom", "Bob", "Tom", "Anna");
